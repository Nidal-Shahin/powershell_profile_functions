function edit-profile { 
    notepad $PROFILE 
}

function save-profile { 
    . $PROFILE 
}

function dir_tree {
    [CmdletBinding()]
    param(
        [string[]]$Exclude,
        [switch]$Interactive
    )

    $defaultExcludes = @('node_modules', '.firebase', 'out', '.next', '.git', 'dist', 'build', '.vs')

    # Handle interactive prompt if -Interactive switch is passed
    if ($Interactive) {
        $userInput = Read-Host "Enter extra folders to exclude (comma-separated) [Default: $($defaultExcludes -join ', ')]"
        if (-not [string]::IsNullOrWhiteSpace($userInput)) {
            $userExcludes = $userInput.Split(',') | ForEach-Object { $_.Trim() } | Where-Object { $_ -ne '' }
            $Exclude += $userExcludes
        }
    }

    # Combine user-provided excludes with defaults
    $finalExcludes = ($defaultExcludes + $Exclude) | Select-Object -Unique

    $outPath = Join-Path $PWD "dir_tree.txt"

    function Get-TreeLines {
        param(
            [string]$Path,
            [string]$Prefix = ''
        )

        # Retrieve items while skipping excluded directory names
        $items = Get-ChildItem -LiteralPath $Path -ErrorAction SilentlyContinue | Where-Object {
            $_.Name -notIn $finalExcludes
        } | Sort-Object PSIsContainer -Descending

        for ($i = 0; $i -lt $items.Count; $i++) {
            $item = $items[$i]
            $isLast = ($i -eq ($items.Count - 1))

            $branch = if ($isLast) { '\-- ' } else { '+-- ' }
            $childPrefix = if ($isLast) { '    ' } else { '|   ' }

            $label = if ($item.PSIsContainer) { "$($item.Name)/" } else { $item.Name }
            "$Prefix$branch$label"

            if ($item.PSIsContainer) {
                Get-TreeLines -Path $item.FullName -Prefix "$Prefix$childPrefix"
            }
        }
    }

    $rootName = Split-Path $PWD -Leaf
    $lines  = @("$rootName/")
    $lines += Get-TreeLines -Path $PWD
    $lines | Set-Content -Path $outPath -Encoding utf8

    $exList = $finalExcludes -join ', '
    Write-Host "Created 'dir_tree.txt' (excluded: $exList)" -ForegroundColor Green
}
